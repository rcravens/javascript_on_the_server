First Step - The Request

MVC Request / Response Lifecycle

When a user enters a URL or clicks a link, the browser will "request" that resource.

  • 👉 Networking (DNS, TLS/SSL, ...) steps.
  • 👉 A connection between the browser (client) and the web application (server) is established.
  • 👉 In our application this is the http.createServer( (req, res) => {} ) function.

HTTP Server

MVC Request / Response Lifecycle


// server.js (partial)

const server = http.createServer(async (req, res) => {
    // url parts (e.g., /users/bob.cravens@tekfoundry.com/edit ==> [users, bob.cravens@tekfoundry.com, edit]
    const url = new URL(req.url, `http://${req.headers.host}`);
    const parts = url.pathname.split("/").filter(Boolean);

    // http verb (e.g., GET, POST, PUT, DELETE...)
    const method = req.method === "POST" && url.searchParams.get("_method")
            ? url.searchParams.get("_method").toUpperCase()
            : req.method;

    try {
        await router.handle(req, res, method, parts);
    } catch (err) {
        console.error("Error handling request:", err);
        res.writeHead(500, {"Content-Type": "text/plain"});
        res.end("Internal Server Error");
    }
});

server.listen(3000, () => {
    console.log("Server running at http://localhost:3000/users");
});

Route Matching

MVC Request / Response Lifecycle


// Router.js (partial)

export default class Router {

    async handle(req, res, method, urlParts) {
        for (const route of this.routes) {
            if (route.method !== method) continue;
            if (route.parts.length !== urlParts.length) continue;

            const params = {};
            let matched = true;

            for (let i = 0; i < route.parts.length; i++) {
                if (route.parts[i].startsWith(":")) {
                    params[route.parts[i].substring(1)] = urlParts[i];
                } else if (route.parts[i] !== urlParts[i]) {
                    matched = false;
                    break;
                }
            }

            if (!matched) continue;

            return ... HANDLER LOGIC ...
        }

        res.writeHead(404, { "Content-Type": "text/plain" });
        res.end("Not Found");
    }
}

Route Handler

MVC Request / Response Lifecycle


// Router.js (partial)

export default class Router {
    async handle(req, res, method, urlParts) {
        for (const route of this.routes) {
            let matched = ... MATCHING CODE ...

            if (!matched) continue;

            const controller = new route.controllerClass();
            if (!controller[route.handlerName])
                throw new Error(`Handler "${route.handlerName}" not found on controller`);

            let body = {};
            if (["POST", "PUT", "PATCH"].includes(method)) {
                body = await Router.parseBody(req);
            }

            return controller[route.handlerName](req, res, params, body);
        }

        res.writeHead(404, { "Content-Type": "text/plain" });
        res.end("Not Found");
    }
}

Request Body Parsing

MVC Request / Response Lifecycle


// Router.js (partial)

export default class Router {
    static parseBody(req) {
        return new Promise((resolve, reject) => {
            let data = '';
            req.on('data', chunk => data += chunk);
            req.on('end', () => {
                try {
                    const contentType = req.headers['content-type'] || '';
                    if (contentType.includes('application/json')) {
                        resolve(JSON.parse(data || '{}'));
                    } else if (contentType.includes('application/x-www-form-urlencoded')) {
                        const params = new URLSearchParams(data);
                        resolve(Object.fromEntries(params.entries()));
                    } else {
                        resolve({});
                    }
                } catch (err) {
                    reject(err);
                }
            });
        });
    }
}

Key Takeaways

  • Every request goes through the same lifecycle.
  • The Server parses the url parts and method and delegates to the Router.
  • The Router matches to a registered route and calls the MVC Logic for that route.
