Why Sessions?

  • HTTP is stateless: each request is independent.
  • Sessions / Session Data:
    • 👉 Stored by your web server (server side).
    • 👉 Data is available only on the server and not on the client.
    • 👉 Track user-specific data on the server by associating it with a session ID stored in a cookie.
    • 👉 Session data should be persistent while user browses website.
    • 👉 Usage: flash message, form validation, cart management...

Simple File-Based Manager


// SessionManager.js (partial)

class SessionManager {
    constructor({storage_dir: storage_dir = null, max_age_seconds: max_age_seconds = 3600} = {}) {
        this.sessions = {};
        this.storage_dir = storage_dir;
        this.max_age_in_milliseconds = max_age_seconds * 1000;
        if (storage_dir) fs.mkdirSync(storage_dir, {recursive: true});
    }

    createSession() {
        const id = randomBytes(16).toString("hex");
        this.sessions[id] = {flash: {}, lastAccess: Date.now()}; // initialize flash container
        if (this.storage_dir) this.saveSessionToFile(id);
        return id;
    }

    saveSessionToFile(id) {
        if (!this.storage_dir || !this.sessions[id]) return;
        const filePath = path.join(this.storage_dir, `${id}.json`);
        fs.writeFileSync(filePath, JSON.stringify(this.sessions[id], null, 2));
    }
}

export const sessionManager = new SessionManager({storage_dir: './data/sessions'});

Sessions

// SessionManager.js (partial)

class SessionManager {

   getSession(id) {
        if (!id) return null;
        if (this.sessions[id]) return this.sessions[id];
        if (this.storage_dir) {
            const filePath = path.join(this.storage_dir, `${id}.json`);
            if (fs.existsSync(filePath)) {
                const data = JSON.parse(fs.readFileSync(filePath));
                this.sessions[id] = data;
                if (!this.sessions[id].flash) this.sessions[id].flash = {};
                return data;
            }
        }
        return null;
    }
}

req.flash = {

        req.session.flash[key] = value;
        if (this.storageDir) this.saveSessionToFile(sessionId);
    },
    get: (key) => {
        const value = req.session.flash[key];
        delete req.session.flash[key];  // remove after reading
        if (this.storageDir) this.saveSessionToFile(sessionId);

Session Cleanup

Simple File-Based Manager


// SessionManager.js (partial)

class SessionManager {

    cleanUpSessions() {
        const now = Date.now();

        // In-memory sessions
        for (const id in this.sessions) {
            const session = this.sessions[id];
            if (session.lastAccess && now - session.lastAccess > this.max_age_in_milliseconds) {
                delete this.sessions[id];
            }
        }

        // Disk sessions
        if (this.storage_dir) {
            const files = fs.readdirSync(this.storage_dir);
            for (const file of files) {
                if (!file.endsWith(".json")) continue;
                const filePath = path.join(this.storage_dir, file);
                const data = JSON.parse(fs.readFileSync(filePath));
                if (data.lastAccess && now - data.lastAccess > this.max_age_in_milliseconds) {
                    fs.unlinkSync(filePath);
                    delete this.sessions[file.replace(".json", "")];
                }
            }
        }
    }
}

Key Takeaways

  • Cookies store client-side data (e.g., session ID).
  • Sessions can store server-side data linked via a cookie (session ID).
  • Production Considerations:
    • 👉 Persistent, redundant, scalable storage solution (e.g., dedicated Redis server)
    • 👉 Consider server-side session storage with libraries like express-session.
